Capstone: sổ cái kép cho ví khách hàng

Mục tiêu: dựng wallets, transfers, ledger_entries để DB tự chặn 4 kiểu tấn công, đối soát được số dư cache, và giao dịch ngược chiều không deadlock.

LÀ GÌ?
Sổ cái kép (double-entry): mỗi lần chuyển tiền ghi 2 bút toán, âm ở ví nguồn, dương ở ví đích, tổng bằng 0. Số dư = tổng bút toán của ví.
TẠI SAO?
Chỉ sửa cột balance thì sai một lần là mất dấu. Có sổ cái thì mọi đồng tiền có nguồn gốc và số dư sai tính lại được.
KHI NÀO KHÔNG?
Bộ đếm lượt xem, điểm không quy ra tiền: một cột n = n + 1 là đủ.
Nhắc lại Bài 14 · 5 dòng
  1. UPDATE, FOR UPDATE giữ khóa dòng tới COMMIT; phiên khác phải chờ (Bài 14 chặng 1).
  2. CHECK (balance >= 0) là lưới cuối: app kiểm tra sai vẫn bị chặn với 23514.
  3. Khóa cùng các dòng theo thứ tự ngược nhau → vòng chờ → 40P01 (Bài 14 chặng 5).
  4. Ai bị hủy do đồng hồ deadlock_timeout (1 s) của từng phiên; đừng viết code dựa vào đó.
  5. Chữa: WHERE id IN (…) ORDER BY id FOR UPDATE, mọi phiên khóa theo id tăng dần.

Sơ đồ và 3 bất biến

3 bất biến
  1. Mỗi giao dịch cân: tổng bút toán của một transfer_id = 0.
  2. Cả hệ thống cân: tổng mọi bút toán = 0; ví SYSTEM_CASH âm đúng bằng tiền mặt đã nạp.
  3. Cache khớp sổ: wallets.balance = tổng bút toán của ví.

DDL gom lại 4 slide: wallets.balance là số dư tính sẵn, tức phi chuẩn hóa (slide 10); transfers.order_ref nằm ở bảng sinh sau (slide 12); version, created_at, updated_at theo table template (slide 19–20); mọi cột tiền là numeric(18,2) (slide 23).

-- #region ddl · Ba bảng, một ràng buộc "mỗi giao dịch tổng bằng 0", một hàm ghi sổ
SET search_path = lab07_modeling;  -- DB mặc định trỏ vào lab03_indexing
SET statement_timeout = '60s'; SET lock_timeout = '5s';
DROP TABLE IF EXISTS ledger_entries, transfers, wallets;
CREATE TABLE wallets (
  id             bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  code           text NOT NULL UNIQUE,
  owner          text NOT NULL,
  balance        numeric(18,2) NOT NULL DEFAULT 0,     -- số dư CACHE (phi chuẩn hóa); sự thật nằm ở ledger_entries
  allow_negative boolean NOT NULL DEFAULT false,       -- chỉ ví hệ thống (tiền mặt nạp vào) được âm
  version        int NOT NULL DEFAULT 0,
  created_at     timestamptz NOT NULL DEFAULT now(),
  updated_at     timestamptz NOT NULL DEFAULT now(),
  CHECK (allow_negative OR balance >= 0)               -- lưới an toàn cuối của Bài 14 (23514)
);
CREATE TABLE transfers (
  id              bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  idempotency_key text NOT NULL UNIQUE,                -- gọi lại cùng request không ghi sổ hai lần
  order_ref       text,                                -- tham chiếu tới đơn hàng (slide 12), NULL nếu là nạp tiền
  amount          numeric(18,2) NOT NULL CHECK (amount > 0),
  created_at      timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE ledger_entries (
  id          bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  transfer_id bigint NOT NULL REFERENCES transfers (id),
  wallet_id   bigint NOT NULL REFERENCES wallets (id),
  amount      numeric(18,2) NOT NULL CHECK (amount <> 0),   -- âm = ghi nợ (tiền ra), dương = ghi có (tiền vào)
  created_at  timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX ON ledger_entries (wallet_id);
CREATE INDEX ON ledger_entries (transfer_id);
-- Kiểm tra "tổng các bút toán của một giao dịch = 0" lúc COMMIT (DEFERRED): code có thể ghi hai bút toán bằng hai câu lệnh riêng,
-- sau câu đầu tổng tạm thời khác 0. (post_transfer ghi cả hai trong MỘT câu INSERT nên kiểm tức thì cũng qua.)
CREATE OR REPLACE FUNCTION ledger_check_balanced() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  IF (SELECT sum(amount) FROM lab07_modeling.ledger_entries WHERE transfer_id = NEW.transfer_id) <> 0 THEN
    RAISE EXCEPTION 'giao dịch % lệch: tổng bút toán khác 0', NEW.transfer_id USING ERRCODE = '23514';
  END IF;
  RETURN NULL;
END $$;
CREATE CONSTRAINT TRIGGER ledger_balanced AFTER INSERT ON ledger_entries
  DEFERRABLE INITIALLY DEFERRED FOR EACH ROW EXECUTE FUNCTION ledger_check_balanced();
-- Sổ cái chỉ được THÊM: sửa sai bằng bút toán đảo (refund), không UPDATE/DELETE bút toán cũ.
CREATE OR REPLACE FUNCTION ledger_append_only() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  RAISE EXCEPTION 'ledger_entries chỉ được thêm, không được %', TG_OP USING ERRCODE = '23001';
END $$;
CREATE TRIGGER ledger_no_update_delete BEFORE UPDATE OR DELETE ON ledger_entries
  FOR EACH ROW EXECUTE FUNCTION ledger_append_only();
-- Ghi một giao dịch chuyển tiền: khóa 2 ví theo thứ tự id (Bài 14), ghi 2 bút toán, cập nhật số dư cache.
CREATE OR REPLACE FUNCTION post_transfer(p_from bigint, p_to bigint, p_amount numeric, p_key text, p_order_ref text DEFAULT NULL)
RETURNS bigint LANGUAGE plpgsql AS $$
DECLARE v_id bigint;
BEGIN
  PERFORM 1 FROM lab07_modeling.wallets WHERE id IN (p_from, p_to) ORDER BY id FOR UPDATE;
  INSERT INTO lab07_modeling.transfers (idempotency_key, order_ref, amount) VALUES (p_key, p_order_ref, p_amount) RETURNING id INTO v_id;
  INSERT INTO lab07_modeling.ledger_entries (transfer_id, wallet_id, amount) VALUES (v_id, p_from, -p_amount), (v_id, p_to, p_amount);
  UPDATE lab07_modeling.wallets SET balance = balance - p_amount, version = version + 1, updated_at = now() WHERE id = p_from;  -- trừ trước
  UPDATE lab07_modeling.wallets SET balance = balance + p_amount, version = version + 1, updated_at = now() WHERE id = p_to;    -- cộng sau
  RETURN v_id;
END $$;
-- #endregion
Làm mẫu · đọc post_transfer từng dòng
  1. PERFORM 1 … ORDER BY id FOR UPDATE: khóa cả hai ví trước, theo id tăng dần, bất kể tiền đi chiều nào (Bài 14).
  2. INSERT INTO transfers: request gửi lần hai đụng UNIQUE idempotency_key.
  3. Hai bút toán -p_amount, +p_amount. post_transfer ghi cả hai trong MỘT câu INSERT nên kiểm tra tức thì cũng qua; trigger để INITIALLY DEFERRED (cộng lúc COMMIT) vì code khác có thể ghi hai bút toán bằng hai câu riêng, như tấn công 2 bên dưới.
  4. Trừ nguồn trước, cộng đích sau: nguồn không đủ thì CHECK dừng ngay ở câu trừ.

Đối soát: cache lệch thì sửa từ sổ

Slide 10: phi chuẩn hóa đổi lấy rủi ro "data consistency". Ai đó cộng tay 5.000 vào ví cửa hàng, không qua sổ; đối soát tìm ra, rồi tính lại số dư từ sổ cái, vì sổ cái là sự thật.

FROM wallets w LEFT JOIN ledger_entries e ON e.wallet_id = w.id GROUP BY w.id ORDER BY w.id;

Vì sao nạn nhân là A?

Quy tắc: mỗi phiên chỉ kiểm tra deadlock một lần, 1 giây (deadlock_timeout) sau khi bắt đầu chờ. B chờ ví 3 từ bước B2; 1 giây sau B kiểm tra, lúc đó A còn ngủ nên chưa có vòng, và B không kiểm tra lại. A tỉnh, xin ví 2, khép vòng; 1001.469 ms sau A kiểm tra, thấy vòng, tự hủy. UPDATE của B treo 5993.849 ms rồi chạy tiếp. Vì vậy A là nạn nhân khi bạn chạy B trong 5 giây đầu lúc A đang ngủ 6 giây; chạy B sau khi A tỉnh thì không có deadlock nào. Ba lần chạy lại (try1–3) đều đúng một 40P01, nạn nhân A (quy tắc Bài 14 chặng 5).

CREATE OR REPLACE FUNCTION refund(p_transfer_id bigint) RETURNS bigint LANGUAGE plpgsql AS $$
DECLARE v_from bigint; v_to bigint; v_amount numeric; v_ref text;
BEGIN
  SELECT max(e.wallet_id) FILTER (WHERE e.amount < 0), max(e.wallet_id) FILTER (WHERE e.amount > 0),
         max(e.amount), max(t.order_ref)
    INTO v_from, v_to, v_amount, v_ref
  FROM lab07_modeling.ledger_entries e JOIN lab07_modeling.transfers t ON t.id = e.transfer_id
  WHERE e.transfer_id = p_transfer_id;
  IF v_amount IS NULL THEN
    RAISE EXCEPTION 'không có giao dịch %', p_transfer_id USING ERRCODE = 'P0002';
  END IF;
  -- đảo chiều; khóa theo id và UNIQUE idempotency_key do post_transfer lo
  RETURN lab07_modeling.post_transfer(v_to, v_from, v_amount, 'refund-' || p_transfer_id, v_ref);
END $$;
